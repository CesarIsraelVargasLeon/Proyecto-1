data <- read.csv("ProstateNHIV2018-2024.csv")
filtro <- data[data$BMI<60 & data$HISPALLP_A>1 & data$SMKCIGST_A<5 & data$DIBEV_A<3 & data$HYPEV_A<3,]
datos <- subset(filtro, select = -c(SMKNOW_A, DIFYRSTC1_A, SMKEV_A))

datos$PROSTCAN_A <- factor(datos$PROSTCAN_A, 
                           levels = c(2, 1), 
                           labels = c("No (Control)", "Sí (Caso)"))

datos$HYPEV_A <- factor(datos$HYPEV_A, 
                     levels = c(2, 1), 
                     labels = c("No", "Sí"))
    
datos$DIBEV_A <- factor(datos$DIBEV_A, 
                     levels = c(2, 1), 
                     labels = c("No", "Sí"))
    
datos$SMKCIGST_A <- factor(datos$SMKCIGST_A, 
                        levels = c(4, 1, 2, 3), 
                        labels = c("Nunca", "Fumador Diario", 
                                   "Fumador Ocasional", "Exfumador"))
    
datos$HISPALLP_A <- factor(datos$HISPALLP_A,
                        levels = c(2, 3),
                        labels = c("Blanco", "Negro"))
